
#include <stdio.h>
#include <limits.h>
#include <float.h>

typedef unsigned short meter_t;

unsigned int consumption(meter_t previous, meter_t current)
{
    return (meter_t)(current - previous);
}

int main(void)
{
    float floatTotal = 0.0f;
    long totalPaise = 0;
    int i;

    printf("sizeof(meter_t) = %zu bytes, maximum reading = %u\n",
           sizeof(meter_t), (unsigned)USHRT_MAX);

    printf("Consumption: %u units\n", consumption(1200, 1530));
    printf("Consumption: %u units\n", consumption(65400, 120));

    printf("\n--- Data Type Information ---\n");
    printf("char: %zu byte(s), range %d to %d\n",
           sizeof(char), CHAR_MIN, CHAR_MAX);
    printf("short: %zu byte(s), range %d to %d\n",
           sizeof(short), SHRT_MIN, SHRT_MAX);
    printf("unsigned short: %zu byte(s), range 0 to %u\n",
           sizeof(unsigned short), (unsigned)USHRT_MAX);
    printf("int: %zu byte(s), range %d to %d\n",
           sizeof(int), INT_MIN, INT_MAX);
    printf("unsigned int: %zu byte(s), maximum %u\n",
           sizeof(unsigned int), UINT_MAX);
    printf("long: %zu byte(s), range %ld to %ld\n",
           sizeof(long), LONG_MIN, LONG_MAX);
    printf("long long: %zu byte(s), range %lld to %lld\n",
           sizeof(long long), LLONG_MIN, LLONG_MAX);

    printf("float: %zu byte(s), FLT_DIG = %d\n",
           sizeof(float), FLT_DIG);
    printf("double: %zu byte(s), DBL_DIG = %d\n",
           sizeof(double), DBL_DIG);

    /* Demonstrate adding 4,000 bills of 10 paise each */
    for (i = 0; i < 4000; i++) {
        floatTotal = floatTotal + 0.10f;
        totalPaise = totalPaise + 10;
    }

    printf("\n--- Adding 4000 Bills ---\n");
    printf("Float total: Rs %.2f\n", floatTotal);
    printf("Integer paise total: Rs %ld.%02ld\n",
           totalPaise / 100, totalPaise % 100);

    printf("\nFLT_MIN = %e\n", FLT_MIN);
    printf("FLT_MAX = %e\n", FLT_MAX);
    printf("DBL_MIN = %e\n", DBL_MIN);
    printf("DBL_MAX = %e\n", DBL_MAX);

    return 0;
}
